# Python ACM 输入输出速查

根据题目灵活选择下面的片段即可。

## 1. 读入

```python
# 逐行（单组 / T 组行数固定）
n = int(input())
a, b = map(int, input().split())
s = input()                          # 整行字符串，含空格也保留

# 一次性读全部 + 指针（最通用、最快，T 组/数组首选）
data = list(map(int, sys.stdin.read().split()))
ptr = 0
t = data[ptr]; ptr += 1
for _ in range(t):
    n = data[ptr]; ptr += 1
    arr = data[ptr: ptr + n]; ptr += n

# 读至 EOF（不告诉组数）
for line in sys.stdin:
    a, b = map(int, line.split())
    print(a + b)

# 更快逐行读（大输入必用，字符串要 strip 掉 \n）
import sys
input = sys.stdin.readline
n = int(input())
s = input().strip()

# 字节级极速读（百万级数据）
data = list(map(int, sys.stdin.buffer.read().split()))

# 字符矩阵 / 逐字符
grid = [list(input().strip()) for _ in range(n)]

# 不定长 / 含空行
while True:
    line = input().strip()
    if line:
        break
# 读到空行结束（多块数据）
while True:
    line = input()
    if not line.strip():
        break

# 图 / 树的边
g = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v); g[v].append(u)

# 十六进制 / 其他进制
x = int(input(), 16)
print(hex(x)[2:])                    # 去 0x 输出
```

## 2. 输出

```python
print(x)                             # 换行
print(x, end=" ")                    # 不换行，空格分隔（一行多值收尾用）
f"{x:.3f}"                           # 保留 3 位小数，不足补 0
f"{x:09d}"                           # 补足 9 位，前补 0
"{:.6f}".format(v)                   # 浮点误差题多保留位数

# 大量输出攒批，比反复 print 快
out = []
out.append(str(ans))
sys.stdout.write("\n".join(out))

# 用空格展开列表（比 end=" " 简洁）
print(*arr)
print(" ".join(map(str, arr)))

# 交互题（实时读写，绝不能用一次性 read，必须 flush）
print(guess, flush=True)
ans = input()

# 格式细节坑
print("Yes" if ok else "No")         # 大小写严格按题目要求
print()                              # 输出空行（组间空行）
# 严格 SPJ：末尾不要多空格，用 join 而非循环 end=" "

# Python 大整数免处理（阶乘 / 大数乘原生支持，无需高精度）
a * b
```

## 3. 常见题型拼装

```python
# T 组一维数组求和
t = int(input())
for _ in range(t):
    n = int(input())
    print(sum(map(int, input().split())))

# T 组二维数组求和
t = int(input())
for _ in range(t):
    n, m = map(int, input().split())
    print(sum(sum(map(int, input().split())) for _ in range(n)))

# 字符串反转（含空格：整行读后过滤）
s = input()
print("".join(ch for ch in reversed(s) if ch != " "))

# 数组一行输出
for i in range(n - 1):
    print(1, end=" ")
print(m - (n - 1))
```
